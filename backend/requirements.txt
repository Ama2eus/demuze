# ── Demuze backend dependencies ───────────────────────────────────────────────

# ⚠  CRITICAL VERSION PINS — DO NOT CHANGE TORCH/TORCHAUDIO VERSIONS
#
#    torchaudio 2.3+ routes ALL audio saving through torchcodec,
#    which is not installed with demucs. This causes the error:
#      "ImportError: TorchCodec is required for save_with_torchcodec"
#
#    torchaudio 2.2.2 is the LAST version that uses soundfile for saving.
#    torch 2.2.2 is the EXACT matching companion version.
#    These pins are mandatory for Demuze to work correctly.
#
# ⚠  DO NOT install this file directly with bare `pip install -r requirements.txt`
#    Use the installer which enforces correct versions and RAM-safe install:
#      ./install_and_run.sh        (macOS / Linux)
#      .\install_and_run.ps1       (Windows)
#
# Manual install (if you know what you're doing):
#   pip install --no-cache-dir torch==2.2.2 torchaudio==2.2.2 \
#       --index-url https://download.pytorch.org/whl/cpu
#   pip install --no-cache-dir fastapi "uvicorn[standard]" python-multipart soundfile demucs
# ─────────────────────────────────────────────────────────────────────────────

# Web framework
fastapi>=0.111.0

torch==2.2.2

demucs>=4.0.1
